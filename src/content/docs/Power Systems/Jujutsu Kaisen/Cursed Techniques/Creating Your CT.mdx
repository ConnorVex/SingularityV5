import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Starting Stats" icon="pencil">
    Copy this onto your sheet:

    ```
    Cursed Technique Type: [Innate or Inherited]
    Cursed Technique Class: [General, Shikigami or Support]
    Cursed Technique Range: Physical or Close Range
    Cursed Technique Power: Below Grade 4
    Cursed Technique Efficiency: Extremely Poor
    Cursed Technique Upgrade Points: 0
    ```

    You can start with **up to 2 Extension Techniques**.
  </Card>
  <Card title="Keep in Mind" icon="warning">
    - CTs follow **one core concept** and never deviate from it. They **can't evolve**, only expand.
    - **Every** use of your CT spends **Cursed Energy**.
    - CTs run on **imagination**, so pick a core you can **expand on**.
  </Card>
</CardGrid>
